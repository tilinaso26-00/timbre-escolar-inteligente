# REQUISITOS PREVIOS (una sola vez, solo para desarrollar)
# Sin Git no se puede clonar la repo; sin Python no corre PlatformIO.
winget install Git.Git
winget install OpenJS.NodeJS.LTS
winget install Python.Python.3.11
winget install Rustlang.Rust.GNU
winget install Microsoft.VisualStudio.2022.BuildTools --override "--add Microsoft.VisualStudio.Workload.NativeDesktop --quiet"

# PlatformIO (usa el Python de arriba)
python --version
pip install platformio

git --version
node --version
npm --version
cargo --version
pio --version
# Nota: JAVA NO hace falta (eso era del Arduino IDE clásico).

# INSTALAR DEPENDENCIAS (una vez)
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm install

# DESARROLLO DIARIO (2 terminales)
# TERMINAL 1 - app web (http://localhost:3000):
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm run dev

# TERMINAL 2 - simulador del ESP32 (ws://localhost:81):
cd /d C:\Users\Robot\Documents\TImbrev1
npm run mock

# O DESARROLLO CON VENTANA DE ESCRITORIO (1 terminal):
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm run tauri:dev

# CONTROLAR EL TIMBRE DESDE EL CELULAR (misma WiFi que la PC)
cd /d C:\Users\Robot\Documents\TImbrev1
npm run build
npm run serve
# -> imprime http://<IP-DE-LA-PC>:4173; abrirla en el navegador del celular
# Si Windows lo bloquea (PowerShell admin):
netsh advfirewall firewall add rule name="Timbre Escolar app" dir=in action=allow protocol=TCP localport=4173
# Parar: Ctrl+C

# BUILD FINAL (GENERAR INSTALADORES .MSI Y .EXE)
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm run tauri:build
# -> src-tauri\target\release\bundle\msi\Timbre Escolar Inteligente_1.0.1_x64_es-ES.msi
# -> src-tauri\target\release\bundle\nsis\Timbre Escolar Inteligente_1.0.1_x64-setup.exe

# CALIDAD DE CÓDIGO (obligatorio antes de guardar)
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm run typecheck
npm run lint

# FIRMWARE (ESP32 por cable USB)
cd /d C:\Users\Robot\Documents\TImbrev1\firmware
C:\Users\Robot\.platformio\penv\Scripts\pio.exe run                # compilar
C:\Users\Robot\.platformio\penv\Scripts\pio.exe run -t upload      # subir (USB)
C:\Users\Robot\.platformio\penv\Scripts\pio.exe run -e esp32dev-ota -t upload   # subir por WiFi (OTA)
C:\Users\Robot\.platformio\penv\Scripts\pio.exe device monitor     # consola serie

# CONFIGURAR WIFI DEL ESP32
# Editar firmware\src\config.h (WIFI_SSID / WIFI_PASSWORD) y volver a subir.

# LIMPIEZA
cd /d C:\Users\Robot\Documents\TImbrev1\app
npm cache clean --force
cd /d C:\Users\Robot\Documents\TImbrev1\app\src-tauri
cargo clean
